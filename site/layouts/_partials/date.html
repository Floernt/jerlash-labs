{{- /* Affiche une date en français (« 31 octobre 2026 », « 1er novembre 2026 ») dans une balise . Accepte une date ou une chaîne. */ -}} {{- with . -}} {{- $d := time.AsTime . -}} {{- $texte := $d | time.Format ":date_long" -}} {{- if eq $d.Day 1 }}{{ $texte = replaceRE `^1 ` "1er " $texte }}{{ end -}} {{ $texte }} {{- end -}}
